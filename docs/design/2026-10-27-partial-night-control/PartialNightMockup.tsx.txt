// #2265 mockups: the control that marks a night partial. Static data, nothing is wired up.
import { View } from 'react-native';
import { AppButton, AppText, Badge, Card, Grid, Icon, IconButton, MetricTile, Stack, SwitchRow } from '../shared/ui';
import { useTheme } from '../shared/theme/ThemeProvider';

type Tile = [label: string, value: string, counted: boolean];
const TILES: Tile[] = [
  ['Deep', '0h 36m', false], ['REM', '0h 42m', false], ['Light', '1h 48m', false], ['Awake', '0h 30m', false],
  ['Efficiency', '84%', false], ['HRV (overnight)', '61 ms', true], ['Average HR', '57 bpm', true],
  ['Lowest HR', '53 bpm', true], ['Breathing rate', '9.1 br/min', true],
];

function Frame({ children }: { children: React.ReactNode }) {
  const { theme } = useTheme();
  return <View style={{ flex: 1, backgroundColor: theme.colors.background, padding: theme.spacing.lg, gap: theme.spacing.md }}>{children}</View>;
}

function NightCard({ marked, header, footer, hint }: { marked: boolean; header?: React.ReactNode; footer?: React.ReactNode; hint?: React.ReactNode }) {
  return (
    <Card tone="sleep">
      <Stack direction="row" align="center" justify="space-between">
        <Stack direction="row" align="center" wrap>
          <AppText variant="eyebrow" tone="muted">2026-10-27</AppText>
          {marked && <Badge label="Partial night" tone="sleep" />}
        </Stack>
        {header}
      </Stack>
      <AppText variant="caption" tone="muted">Oura · Ring</AppText>
      <AppText variant="title">{marked ? '3h 5m recorded' : '3h 5m'}</AppText>
      <AppText variant="caption" tone="muted">4:23 am → 8:03 am</AppText>
      {marked && (
        <AppText variant="caption" tone="muted">
          Kept out of your sleep score and averages. Heart rate, HRV and breathing still count.
        </AppText>
      )}
      {hint}
      <Grid columns={2}>
        {TILES.map(([label, value, counted]) => (
          <View key={label} style={{ opacity: marked && !counted ? 0.4 : 1 }}>
            <MetricTile label={label} value={value} tone="sleep" />
          </View>
        ))}
      </Grid>
      {footer}
    </Card>
  );
}

function Sheet({ children }: { children: React.ReactNode }) {
  const { theme } = useTheme();
  return (
    <View style={{ position: 'absolute', left: 0, right: 0, bottom: 0, padding: theme.spacing.lg, backgroundColor: theme.colors.raised, borderTopLeftRadius: 20, borderTopRightRadius: 20, gap: theme.spacing.md, paddingBottom: theme.spacing.xl }}>
      {children}
    </View>
  );
}

export function MockupApp({ screen }: { screen: string }) {
  const { theme } = useTheme();
  const toggle = (on: boolean) => (
    <View style={{ borderTopWidth: 1, borderTopColor: theme.colors.border, paddingTop: theme.spacing.md, gap: theme.spacing.xs }}>
      <SwitchRow label="The ring was off for part of this night" value={on} onValueChange={() => {}} />
      <AppText variant="caption" tone="muted">
        {on ? 'Turn off to count the whole night again.' : 'Keeps hours, efficiency and bedtime out of your scores and averages.'}
      </AppText>
    </View>
  );
  if (screen === 'a-off') return <Frame><NightCard marked={false} footer={toggle(false)} /></Frame>;
  if (screen === 'a-on') return <Frame><NightCard marked footer={toggle(true)} /></Frame>;
  if (screen === 'b-menu')
    return (
      <Frame>
        <NightCard marked={false} header={<IconButton icon="more" label="Night options" onPress={() => {}} />} />
        <View style={{ position: 'absolute', top: 64, right: 24, width: 230, backgroundColor: theme.colors.raised, borderRadius: 12, padding: theme.spacing.md, gap: theme.spacing.sm }}>
          <AppText variant="body">Mark as partial night</AppText>
          <AppText variant="body" tone="muted">Log bedtime by hand</AppText>
        </View>
      </Frame>
    );
  if (screen === 'b-confirm')
    return (
      <Frame>
        <NightCard marked={false} header={<IconButton icon="more" label="Night options" onPress={() => {}} />} />
        <Sheet>
          <AppText variant="title">Mark as a partial night?</AppText>
          <AppText variant="body" tone="muted">
            Use this when the ring came off or went on late. Hours, efficiency, time to fall asleep and bedtime stay out of your sleep score and averages. Heart rate, HRV and breathing still count. You can undo it any time.
          </AppText>
          <AppButton title="Mark as partial night" onPress={() => {}} />
          <AppButton title="Cancel" variant="surface" onPress={() => {}} />
        </Sheet>
      </Frame>
    );
  if (screen === 'b-marked')
    return (
      <Frame>
        <NightCard marked header={<IconButton icon="more" label="Night options" onPress={() => {}} />} />
        <Sheet>
          <AppText variant="body">Partial night marked</AppText>
          <AppButton title="Count the whole night again" variant="surface" onPress={() => {}} />
        </Sheet>
      </Frame>
    );
  if (screen === 'c-prompt')
    return (
      <Frame>
        <NightCard
          marked={false}
          hint={
            <Card tone="brand">
              <Stack direction="row" gap="sm" align="center"><Icon name="info" tone="muted" /><AppText variant="body">Short night. Was the ring off for part of it?</AppText></Stack>
              <Stack direction="row" gap="sm">
                <AppButton title="Yes, mark partial" onPress={() => {}} />
                <AppButton title="No" variant="surface" onPress={() => {}} />
              </Stack>
            </Card>
          }
        />
      </Frame>
    );
  return <Frame><NightCard marked={false} /></Frame>;
}
